## mysql

SET GLOBAL long_query_time = 0;
SET GLOBAL slow_query_log = ON;
SET GLOBAL slow_query_log_file = '/var/lib/mysql/slow-query-manual.log';

SET GLOBAL slow_query_log = OFF;
SET GLOBAL long_query_time = 10;

// スロークエリ設定確認
docker compose -f webapp/compose.yml exec mysql \
 mysql -uroot -proot -e "
SHOW GLOBAL VARIABLES
WHERE Variable_name IN (
'slow_query_log',
'slow_query_log_file',
'long_query_time',
'log_output'
);
"

// アプリの再起動
docker compose restart app

## tools/alp

// ベンチマーク実行前にログを空にする
docker compose -f webapp/compose.yml exec nginx \
 sh -c ': > /var/log/nginx/isucon_access.log'

// ベンチ後、レスポンスタイム合計が大きい順にtop10まで表示
docker compose -f webapp/compose.yml run --rm accesslog-analyzer \
 ltsv \
 --file /var/log/nginx/isucon_access.log \
 --sort sum \
 --reverse \
 --output count,method,uri,sum,avg,p95 \
 --format tsv |
awk 'NR <= 11' |
column -t -s $'\t'
